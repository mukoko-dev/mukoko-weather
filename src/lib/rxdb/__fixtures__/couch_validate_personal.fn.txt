  // A design document is written by a server admin only. This is also how this
  // function replaces itself: the old one runs against the new design document.
  if (typeof id === "string" && id.indexOf("_design/") === 0) {
    if (userCtx && userCtx.roles && inList(userCtx.roles, "_admin")) { return; }
    reject("only a server admin writes a design document");
  }

  // 1. Owner only. There is no admin exception for data: the gateway holds the only
  // admin credential, and it never writes a person's documents.
  if (!userCtx || !userCtx.name) { throw({unauthorized: "sign in"}); }
  if (userCtx.db !== DB_PREFIX + userCtx.name) { reject("this database belongs to someone else"); }

  // 2. A listed type, and the id starts with it.
  var type = newDoc.type;
  if (typeof id !== "string" || typeof type !== "string" || !has(SCHEMAS, type)) { reject("unknown document type"); }
  if (id.indexOf(type + ":") !== 0) { reject("_id must start with the type and a colon"); }

  // 6. Size, in bytes of UTF-8 JSON.
  var bytes;
  try { bytes = unescape(encodeURIComponent(JSON.stringify(newDoc))).length; }
  catch (err) { reject("the document is not valid Unicode text"); }
  if (bytes > MAX_BYTES) { reject("the document is larger than the limit"); }

  // 3. The type's schema.
  var why = check(SCHEMAS[type], newDoc, "doc");
  if (why !== null) { reject(why); }

  // 4. A natural key equals the field it stands for.
  var keyField = KEYS[type];
  if (keyField && has(newDoc, keyField) && String(newDoc[keyField]) !== id.substring(type.length + 1)) {
    reject("_id does not match " + keyField);
  }

  // 5. An envelope is a JWE of the profile. The schema has fixed its members; here the
  // protected header is decoded and held to alg, enc and kid, and the aad to this _id.
  if (has(newDoc, "envelope")) {
    var raw = b64uDecode(newDoc.envelope.protected), header = null;
    try { header = JSON.parse(decodeURIComponent(escape(raw))); } catch (err2) { header = null; }
    if (header === null) { reject("doc.envelope.protected is not a JSON header"); }
    why = check(PROTECTED, header, "doc.envelope.protected");
    if (why !== null) { reject(why); }
    if (newDoc.envelope.aad !== b64uEncode(id)) { reject("doc.envelope.aad does not name this document"); }
  }

  // 7. A far-future updatedAt would win every conflict until then.
  if (Date.parse(newDoc.updatedAt) > Date.now() + FUTURE_MS) { reject("updatedAt is too far ahead"); }
}
